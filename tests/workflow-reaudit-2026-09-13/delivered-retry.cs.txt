public static class GeneratedTranslations
{
	public static void AfterInjection(LoadedLanguage __instance)
	{
		Apply(DefDatabase<ThingDef>.AllDefsListForReading, __instance, delegate(string key, string chunk)
		{
			//IL_0007: Unknown result type (might be due to invalid IL or missing references)
			//IL_000c: Unknown result type (might be due to invalid IL or missing references)
			//IL_0011: Unknown result type (might be due to invalid IL or missing references)
			TaggedString val = TranslatorFormattedStringExtensions.Translate(key, NamedArgumentUtility.Named((object)chunk, "CHUNK"));
			return ((object)(TaggedString)(ref val)).ToString();
		});
	}

	public static void Apply(IEnumerable<ThingDef> defs, LoadedLanguage language, Func<string, string, string> translate)
	{
		if (language.folderName == "English")
		{
			return;
		}
		HashSet<string> hashSet = new HashSet<string>(StringComparer.Ordinal);
		foreach (DefInjectionPackage defInjection in language.defInjections)
		{
			if (!(defInjection.defType == typeof(ThingDef)))
			{
				continue;
			}
			foreach (DefInjection value2 in defInjection.injections.Values)
			{
				if (value2.injected && !value2.isPlaceholder)
				{
					hashSet.Add(value2.normalizedPath ?? value2.path);
				}
			}
		}
		foreach (ThingDef def in defs)
		{
			if (((BuildableDef)def).costList == null || ((BuildableDef)def).costList.Count != 1)
			{
				continue;
			}
			ThingDef thingDef = ((BuildableDef)def).costList[0].thingDef;
			if (thingDef == null || string.IsNullOrEmpty(((Def)thingDef).label))
			{
				continue;
			}
			string[] array = new string[3] { "LargePot", "Plinth", "ChunkStorage" };
			foreach (string text in array)
			{
				if (((Def)def).defName != "ASNeolithic" + text + ((Def)thingDef).defName)
				{
					continue;
				}
				string[] array2 = new string[2] { "label", "description" };
				foreach (string text2 in array2)
				{
					if (hashSet.Contains(((Def)def).defName + "." + text2))
					{
						continue;
					}
					string text3 = "ASNeolithic.Generated." + text + "." + text2;
					if (language.keyedReplacements.TryGetValue(text3, out var value) && !value.isPlaceholder && !string.IsNullOrWhiteSpace(value.value))
					{
						string text4 = translate(text3, ((Def)thingDef).label);
						if (text2 == "label")
						{
							((Def)def).label = text4;
						}
						else
						{
							((Def)def).description = text4;
						}
					}
				}
				((Def)def).ClearCachedData();
				break;
			}
		}
	}
}
